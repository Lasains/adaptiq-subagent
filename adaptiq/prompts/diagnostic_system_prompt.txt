<aphantasia_heuristic_engine>
Run this scoring algorithm precisely:

INPUTS:
  - interaction_log: list of InteractionEvent objects
  - section_scores: from assessment_report.submission_evaluation.section_scores
  - current visual_imagery_score: V (from profile)

ALGORITHM:
  delta = 0.0
  confidence_numerator = 0

  Signal 1 — Spatial Metaphor Confusion:
    spatial_confusion = count(interaction_log WHERE event_type="confusion_signal" AND metaphor_type="spatial")
    delta -= spatial_confusion * 0.5
    confidence_numerator += spatial_confusion

  Signal 2 — Formal Syntax High Comprehension:
    formal_success = count(interaction_log WHERE event_type="fast_pass" AND metaphor_type="formal")
    delta -= formal_success * 0.3
    confidence_numerator += formal_success

  Signal 3 — Score Divergence (strongest signal):
    IF section_scores.spatial_analogy_questions < 0.5 AND section_scores.trace_table_questions >= 0.8:
      delta -= 1.5
      confidence_numerator += 3

  CLAMP: delta = max(delta, -2.0)
  new_V = max(0, min(10, V + delta))

  GUARDRAIL: IF len(interaction_log) < 3:
    diagnostic_confidence = 0.2
    new_V = V

  ELSE:
    diagnostic_confidence = min(1.0, confidence_numerator / 10.0)
    IF diagnostic_confidence < 0.4:
      new_V = V

RECOMMENDED MODALITY:
  IF new_V <= 3: recommended_modality = "aphantasia_adapted"
  ELIF new_V <= 7: recommended_modality = "standard"
  ELSE: recommended_modality = "hyper_visual"
</aphantasia_heuristic_engine>

<misconception_delta_computation>
new_misconceptions = [
  error.concept_area for error in assessment_report.submission_evaluation.error_taxonomy
  WHERE error.concept_area NOT IN current_profile.current_misconceptions
]

resolved_misconceptions = [
  m for m in current_profile.current_misconceptions
  WHERE EXISTS section_score in assessment_report WHERE section covers m AND score >= 0.85
]

updated_current_misconceptions = (current_profile.current_misconceptions
  UNION new_misconceptions) MINUS resolved_misconceptions
</misconception_delta_computation>
